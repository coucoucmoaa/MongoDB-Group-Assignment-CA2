// 1. Database
// getSiblingDB is used instead of "use" so the script also works with load()
db = db.getSiblingDB("cityTrashSystem");

// dates are generated relative to today so time-based queries always return results
var DAY = 24 * 60 * 60 * 1000;
var daysAgo = (d) => new Date(Date.now() - d * DAY);
var tsDaysAgo = (d) => Timestamp({ t: Math.floor((Date.now() - d * DAY) / 1000), i: 1 });


// 2. recyclingBins collection
// MongoDB is schema-less, so the schema from Part 1 is enforced with a $jsonSchema validator.
// _id is generated automatically (ObjectId). fillLevel is a percentage, capacity is in litres.
db.recyclingBins.drop();

db.createCollection("recyclingBins", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["binId", "binType", "status", "isOverflow", "fillLevel", "capacity", "area", "location"],
      properties: {
        binId: { bsonType: "number" },
        binType: { bsonType: "string" },
        status: { bsonType: "string" },
        isOverflow: { bsonType: "bool" },
        fillLevel: { bsonType: "number", minimum: 0, maximum: 100 },
        capacity: { bsonType: "number", minimum: 0 },
        lastCollectedAt: { bsonType: "date" },
        acceptedMaterials: { bsonType: "array", items: { bsonType: "string" } },
        area: { bsonType: "string" },
        location: {
          bsonType: "object",
          required: ["type", "coordinates"],
          properties: {
            type: { enum: ["Point"] },
            coordinates: { bsonType: "array", minItems: 2, maxItems: 2, items: { bsonType: "number" } }
          }
        },
        maintenanceHistory: {
          bsonType: "array",
          items: {
            bsonType: "object",
            required: ["maintenancePurpose", "maintenanceDate"],
            properties: {
              maintenancePurpose: { bsonType: "string" },
              description: { bsonType: "string" },
              maintenanceDate: { bsonType: "timestamp" }
            }
          }
        }
      }
    }
  },
  validationLevel: "strict",
  validationAction: "error"
});

// indexes: unique binId, 2dsphere for geo queries (nearest bin...), area for grouping by arrondissement
db.recyclingBins.createIndex({ binId: 1 }, { unique: true });
db.recyclingBins.createIndex({ location: "2dsphere" });
db.recyclingBins.createIndex({ area: 1 });

// 4. wasteReports collection
//   - area (String): arrondissement of the report, to group reports by zone
//   - resolvedAt (Date or null): when the issue was fixed, to compute resolution times
//   - binId (Number, optional): reference to recyclingBins.binId for overflowing / missed collection reports
// reportedAt is stored as a Date so it can be used in date range queries and date arithmetic.
db.wasteReports.drop();

db.createCollection("wasteReports", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["category", "wasteType", "description", "status", "reportedAt", "area", "location", "reporter"],
      properties: {
        category: { enum: ["illegalDumping", "missedCollection", "overflowingBin", "hazardousWaste"] },
        wasteType: { bsonType: "string" },
        description: { bsonType: "string" },
        status: { bsonType: "string" },
        estimatedWeight: { bsonType: "number", minimum: 0 },
        reportedAt: { bsonType: "date" },
        resolvedAt: { bsonType: ["date", "null"] },
        photos: { bsonType: "array", items: { bsonType: "string" } },
        area: { bsonType: "string" },
        binId: { bsonType: "number" },
        location: {
          bsonType: "object",
          required: ["type", "coordinates"],
          properties: {
            type: { enum: ["Point"] },
            coordinates: { bsonType: "array", minItems: 2, maxItems: 2, items: { bsonType: "number" } }
          }
        },
        reporter: {
          bsonType: "object",
          required: ["username", "isVerifiedUser"],
          properties: {
            username: { bsonType: "string" },
            isVerifiedUser: { bsonType: "bool" }
          }
        }
      }
    }
  },
  validationLevel: "strict",
  validationAction: "error"
});

// indexes: category + area for hotspot queries, reportedAt for date ranges, 2dsphere for geo queries,
// reporter.username for citizen statistics, binId for the join with recyclingBins,
// and a text index on description/wasteType for keyword search
db.wasteReports.createIndex({ category: 1, area: 1 });
db.wasteReports.createIndex({ reportedAt: -1 });
db.wasteReports.createIndex({ location: "2dsphere" });
db.wasteReports.createIndex({ "reporter.username": 1 });
db.wasteReports.createIndex({ binId: 1 });
db.wasteReports.createIndex({ description: "text", wasteType: "text" });

// PART 3 - Querying

// Query 1 - Unresolved reports (reported or in progress), most recent first
// basic select with a filter on a list of values ($in)
// Result: 15 reports
db.wasteReports.find(
  { status: { $in: ["reported", "inProgress"] } },
  { _id: 0, category: 1, area: 1, status: 1, reportedAt: 1 }
).sort({ reportedAt: -1 });

// Query 2 - Overflowing bins, the ones collected the longest time ago first
// filter on a boolean + sort on a date + projection
// Result: 4 bins - 1001, 1006, 1004, 1003
db.recyclingBins.find(
  { isOverflow: true },
  { _id: 0, binId: 1, binType: 1, area: 1, fillLevel: 1, lastCollectedAt: 1 }
).sort({ lastCollectedAt: 1 });

// Query 3 - Heaviest reports, heaviest first comparison operator ($gt)
// on a double + sort
// Result: 7 reports
db.wasteReports.find(
  { estimatedWeight: { $gt: 30 } },
  { _id: 0, category: 1, wasteType: 1, estimatedWeight: 1, area: 1 }
).sort({ estimatedWeight: -1 });

// Query 4 - Bins that accept glass bottles
// Result: 3 bins - 1001 (Paris 11e), 1005 (Paris 18e), 1011 (Paris 15e)
db.recyclingBins.find(
  { acceptedMaterials: "glass bottles" },
  { _id: 0, binId: 1, binType: 1, area: 1, acceptedMaterials: 1 }
);

// Query 5 - Reports with at least 2 photos, with the number of photos
// Result: 4 reports - 3 with 2 photos (Paris 18e, 11e, 13e) and 1 with 3 photos (Paris 20e)
db.wasteReports.find(
  { $expr: { $gte: [{ $size: "$photos" }, 2] } },
  { _id: 0, category: 1, area: 1, description: 1, nbPhotos: { $size: "$photos" } }
);

// Query 6 - Bins that had a repair during the last 30 days
// Result: 2 bins - 1003 (Paris 11e) and 1010 (Paris 13e)
db.recyclingBins.find(
  { maintenanceHistory: { $elemMatch: { maintenancePurpose: "repair", maintenanceDate: { $gte: tsDaysAgo(30) } } } },
  { _id: 0, binId: 1, area: 1, status: 1, maintenanceHistory: 1 }
);

// Query 7 - Keyword search: reports mentioning a bad smell
// $text uses the text index created on description and wasteType, results sorted by relevance score
// Result: 3 reports (Paris 19e, 11e and 18e)
db.wasteReports.find(
  { $text: { $search: "smell" } },
  { _id: 0, category: 1, area: 1, description: 1, score: { $meta: "textScore" } }
).sort({ score: { $meta: "textScore" } });

// Query 8 - Bins less than 2 km away from Place de la Republique, nearest first
// geospatial query on the GPS location using the 2dsphere index ($near sorts by distance)
// Result: 4 bins - 1008, 1001, 1003 and 1002
db.recyclingBins.find(
  {
    location: {
      $near: {
        $geometry: { type: "Point", coordinates: [2.3631, 48.8674] },
        $maxDistance: 2000
      }
    }
  },
  { _id: 0, binId: 1, binType: 1, area: 1, status: 1 }
);

// Query 9 - Illegal dumping hotspots: top 5 areas with the most unresolved illegal dumping reports in the past month
// aggregation pipeline: $match (filter) -> $group (count per area) -> $sort -> $limit
// Result: Paris 18e (3 reports), Paris 11e (2), then Paris 10e, 19e and 20e (1 each)
db.wasteReports.aggregate([
  {
    $match: {
      category: "illegalDumping",
      status: { $in: ["reported", "inProgress"] },
      reportedAt: { $gte: daysAgo(30) }
    }
  },
  {
    $group: {
      _id: "$area",
      unresolvedReports: { $sum: 1 },
      totalEstimatedWeight: { $sum: "$estimatedWeight" }
    }
  },
  { $sort: { unresolvedReports: -1, _id: 1 } },
  { $limit: 5 },
  { $project: { _id: 0, area: "$_id", unresolvedReports: 1, totalEstimatedWeight: 1 } }
]);

// Query 10 - Bin overflow tracking: bins with the most overflow alerts reported by citizens over the last 3 weeks
// $group counts the alerts per binId, then $lookup joins recyclingBins (cross-reference) to get the bin details
// Result: bin 1003 (3 alerts), 1001 (2), 1006 (2), 1004 (1)
db.wasteReports.aggregate([
  { $match: { category: "overflowingBin", reportedAt: { $gte: daysAgo(21) } } },
  { $group: { _id: "$binId", overflowAlerts: { $sum: 1 }, lastAlert: { $max: "$reportedAt" } } },
  { $sort: { overflowAlerts: -1, _id: 1 } },
  { $lookup: { from: "recyclingBins", localField: "_id", foreignField: "binId", as: "bin" } },
  { $unwind: "$bin" },
  {
    $project: {
      _id: 0,
      binId: "$_id",
      overflowAlerts: 1,
      lastAlert: 1,
      area: "$bin.area",
      binType: "$bin.binType",
      fillLevel: "$bin.fillLevel",
      lastCollectedAt: "$bin.lastCollectedAt"
    }
  }
]);

// Query 11 - Citizen engagement: most active citizens with verified reports over the last 6 months
// $match on a field of the embedded document reporter, $addToSet lists the different categories reported
// Result: marie_dupont (6 reports), sofiane75 (4), then camille_r, julien.m and leaD (3 each)
db.wasteReports.aggregate([
  { $match: { "reporter.isVerifiedUser": true, reportedAt: { $gte: daysAgo(182) } } },
  { $group: { _id: "$reporter.username", reportsSubmitted: { $sum: 1 }, categories: { $addToSet: "$category" } } },
  { $sort: { reportsSubmitted: -1, _id: 1 } },
  { $limit: 5 },
  { $project: { _id: 0, username: "$_id", reportsSubmitted: 1, categories: 1 } }
]);

// Query 12 - Collection response efficiency: average resolution time of missed collection reports per area, fastest first
// $subtract between two dates gives milliseconds, divided by 3 600 000 to get hours
// Result: Paris 19e (12 h), Paris 11e (36 h), Paris 18e (72 h), Paris 15e (96 h)
db.wasteReports.aggregate([
  { $match: { category: "missedCollection", resolvedAt: { $ne: null } } },
  { $project: { area: 1, resolutionHours: { $divide: [{ $subtract: ["$resolvedAt", "$reportedAt"] }, 3600000] } } },
  { $group: { _id: "$area", avgResolutionHours: { $avg: "$resolutionHours" }, resolvedReports: { $sum: 1 } } },
  { $sort: { avgResolutionHours: 1 } },
  { $project: { _id: 0, area: "$_id", avgResolutionHours: { $round: ["$avgResolutionHours", 1] }, resolvedReports: 1 } }
]);

// Query 13 - Waste collected impact: total weight of waste collected (resolved reports) per type of waste
// adaptation of the "Community Cleanup Impact" example: $group by wasteType with $sum on estimatedWeight
// Result: bulky items 90 kg (2 reports), glass 67 kg (3), household waste 38.5 kg (4), plastic 27 kg (2), organic 12 kg (1), batteries 3.5 kg (1)
db.wasteReports.aggregate([
  { $match: { status: "resolved" } },
  { $group: { _id: "$wasteType", totalWeightKg: { $sum: "$estimatedWeight" }, resolvedReports: { $sum: 1 } } },
  { $sort: { totalWeightKg: -1 } },
  { $project: { _id: 0, wasteType: "$_id", totalWeightKg: 1, resolvedReports: 1 } }
]);

// Query 14 - Maintenance activity: number of interventions per maintenance purpose and the bins concerned
// $unwind creates one document per element of the maintenanceHistory array so the embedded documents can be grouped
// Result: inspection 3 (bins 1012, 1004, 1006), repair 3 (bins 1001, 1003, 1010), cleaning 2 (bins 1003, 1004),
// replacement 1 (bin 1004), sensorCalibration 1 (bin 1009)
db.recyclingBins.aggregate([
  { $unwind: "$maintenanceHistory" },
  { $group: { _id: "$maintenanceHistory.maintenancePurpose", interventions: { $sum: 1 }, bins: { $addToSet: "$binId" } } },
  { $sort: { interventions: -1, _id: 1 } },
  { $project: { _id: 0, maintenancePurpose: "$_id", interventions: 1, bins: 1 } }
]);

// Query 15 - Bin status per area: number of bins, overflowing bins and average fill level, most critical areas first
// $cond counts only the bins where isOverflow is true
// Result: Paris 11e (3 bins, 2 overflowing, 95% average fill), Paris 18e (2 bins, 1 overflowing, 80% average fill),
// Paris 19e (2 bins, 1 overflowing, 72.8% average fill), Paris 15e (1 bin, 0 overflowing, 75% average fill),
// Paris 10e (1 bin, 0 overflowing, 30% average fill), Paris 5e (1 bin, 0 overflowing, 20% average fill),
// Paris 13e (1 bin, 0 overflowing, 10% average fill), Paris 20e (1 bin, 0 overflowing, 0% average fill)
db.recyclingBins.aggregate([
  { $group: { _id: "$area", totalBins: { $sum: 1 }, overflowingBins: { $sum: { $cond: ["$isOverflow", 1, 0] } }, avgFillLevel: { $avg: "$fillLevel" } } },
  { $sort: { overflowingBins: -1, avgFillLevel: -1 } },
  { $project: { _id: 0, area: "$_id", totalBins: 1, overflowingBins: 1, avgFillLevel: { $round: ["$avgFillLevel", 1] } } }
]);
