// 1. Database
// getSiblingDB is used instead of "use" so the script also works with load()
db = db.getSiblingDB("cityTrashSystem");

// dates are generated relative to today so time-based queries always return results
var DAY = 24 * 60 * 60 * 1000;
var daysAgo = (d) => new Date(Date.now() - d * DAY);
var tsDaysAgo = (d) => Timestamp({ t: Math.floor((Date.now() - d * DAY) / 1000), i: 1 });


// 2. recyclingBins collection
// MongoDB is schema-less, so the schema from Part 1 is enforced with a $jsonSchema validator.
// _id is generated automatically (ObjectId). fillLevel is a percentage, capacity is in litres.
db.recyclingBins.drop();

db.createCollection("recyclingBins", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["binId", "binType", "status", "isOverflow", "fillLevel", "capacity", "area", "location"],
      properties: {
        binId: { bsonType: "number" },
        binType: { bsonType: "string" },
        status: { bsonType: "string" },
        isOverflow: { bsonType: "bool" },
        fillLevel: { bsonType: "number", minimum: 0, maximum: 100 },
        capacity: { bsonType: "number", minimum: 0 },
        lastCollectedAt: { bsonType: "date" },
        acceptedMaterials: { bsonType: "array", items: { bsonType: "string" } },
        area: { bsonType: "string" },
        location: {
          bsonType: "object",
          required: ["type", "coordinates"],
          properties: {
            type: { enum: ["Point"] },
            coordinates: { bsonType: "array", minItems: 2, maxItems: 2, items: { bsonType: "number" } }
          }
        },
        maintenanceHistory: {
          bsonType: "array",
          items: {
            bsonType: "object",
            required: ["maintenancePurpose", "maintenanceDate"],
            properties: {
              maintenancePurpose: { bsonType: "string" },
              description: { bsonType: "string" },
              maintenanceDate: { bsonType: "timestamp" }
            }
          }
        }
      }
    }
  },
  validationLevel: "strict",
  validationAction: "error"
});

// indexes: unique binId, 2dsphere for geo queries (nearest bin...), area for grouping by arrondissement
db.recyclingBins.createIndex({ binId: 1 }, { unique: true });
db.recyclingBins.createIndex({ location: "2dsphere" });
db.recyclingBins.createIndex({ area: 1 });

// 4. wasteReports collection
//   - area (String): arrondissement of the report, to group reports by zone
//   - resolvedAt (Date or null): when the issue was fixed, to compute resolution times
//   - binId (Number, optional): reference to recyclingBins.binId for overflowing / missed collection reports
// reportedAt is stored as a Date so it can be used in date range queries and date arithmetic.
db.wasteReports.drop();

db.createCollection("wasteReports", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["category", "wasteType", "description", "status", "reportedAt", "area", "location", "reporter"],
      properties: {
        category: { enum: ["illegalDumping", "missedCollection", "overflowingBin", "hazardousWaste"] },
        wasteType: { bsonType: "string" },
        description: { bsonType: "string" },
        status: { bsonType: "string" },
        estimatedWeight: { bsonType: "number", minimum: 0 },
        reportedAt: { bsonType: "date" },
        resolvedAt: { bsonType: ["date", "null"] },
        photos: { bsonType: "array", items: { bsonType: "string" } },
        area: { bsonType: "string" },
        binId: { bsonType: "number" },
        location: {
          bsonType: "object",
          required: ["type", "coordinates"],
          properties: {
            type: { enum: ["Point"] },
            coordinates: { bsonType: "array", minItems: 2, maxItems: 2, items: { bsonType: "number" } }
          }
        },
        reporter: {
          bsonType: "object",
          required: ["username", "isVerifiedUser"],
          properties: {
            username: { bsonType: "string" },
            isVerifiedUser: { bsonType: "bool" }
          }
        }
      }
    }
  },
  validationLevel: "strict",
  validationAction: "error"
});

// indexes: category + area for hotspot queries, reportedAt for date ranges, 2dsphere for geo queries,
// reporter.username for citizen statistics, binId for the join with recyclingBins,
// and a text index on description/wasteType for keyword search
db.wasteReports.createIndex({ category: 1, area: 1 });
db.wasteReports.createIndex({ reportedAt: -1 });
db.wasteReports.createIndex({ location: "2dsphere" });
db.wasteReports.createIndex({ "reporter.username": 1 });
db.wasteReports.createIndex({ binId: 1 });
db.wasteReports.createIndex({ description: "text", wasteType: "text" });

// PART 3 - Querying

// Query 1 - Unresolved reports (reported or in progress), most recent first
// basic select with a filter on a list of values ($in)
// Result: 15 reports
db.wasteReports.find(
  { status: { $in: ["reported", "inProgress"] } },
  { _id: 0, category: 1, area: 1, status: 1, reportedAt: 1 }
).sort({ reportedAt: -1 });

// Query 2 - Overflowing bins, the ones collected the longest time ago first
// filter on a boolean + sort on a date + projection
// Result: 4 bins - 1001, 1006, 1004, 1003
db.recyclingBins.find(
  { isOverflow: true },
  { _id: 0, binId: 1, binType: 1, area: 1, fillLevel: 1, lastCollectedAt: 1 }
).sort({ lastCollectedAt: 1 });

// Query 3 - Heaviest reports, heaviest first comparison operator ($gt)
// on a double + sort
// Result: 7 reports
db.wasteReports.find(
  { estimatedWeight: { $gt: 30 } },
  { _id: 0, category: 1, wasteType: 1, estimatedWeight: 1, area: 1 }
).sort({ estimatedWeight: -1 });
